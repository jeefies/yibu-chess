cmake_minimum_required(VERSION 3.22.1)
project(yibu_stockfish LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SF "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/stockfish/src")
file(GLOB_RECURSE SF_SOURCES CONFIGURE_DEPENDS "${SF}/*.cpp")
list(FILTER SF_SOURCES EXCLUDE REGEX "/main\\.cpp$")
add_library(yibu_stockfish SHARED bridge.cpp ${SF_SOURCES})
set_property(GLOBAL PROPERTY JOB_POOLS sf_compile=2)
set_property(TARGET yibu_stockfish PROPERTY JOB_POOL_COMPILE sf_compile)
target_include_directories(yibu_stockfish PRIVATE "${SF}")
target_compile_definitions(yibu_stockfish PRIVATE NNUE_EMBEDDING_OFF NDEBUG USE_NEON IS_64BIT)
target_compile_options(yibu_stockfish PRIVATE -O3 -fexceptions -frtti)
target_link_options(yibu_stockfish PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(yibu_stockfish android log)
